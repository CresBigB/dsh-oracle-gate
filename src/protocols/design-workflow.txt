// ── 清单路径索引：P4 用它核对 reuses 是不是编的 ───────────────────────
const knownPaths = new Set()
for (const line of inventory.split('\n')) {
  // 清单行格式：`- <path>:<line> — <kind> — <name>`
  const match = line.match(/^- (\S+):\d+ — /)
  if (match !== null) knownPaths.add(match[1].replace(/\\/g, '/'))
}

function normalizePath(raw) {
  return String(raw ?? '').replace(/\\/g, '/').trim()
}

/** reuses 条目形如 `path:line — 用途`，取出 path 部分。 */
function reusePathOf(entry) {
  const match = String(entry).match(/^(.+?):\d+/)
  return match === null ? null : normalizePath(match[1])
}

/**
 * 路径是否真的在清单里。
 * 宽容处理绝对路径与相对路径的差异：agent 有时会补上项目前缀。
 */
function isKnownPath(candidate) {
  const path = normalizePath(candidate)
  if (path === '') return false
  if (knownPaths.has(path)) return true
  for (const known of knownPaths) {
    if (path === known || path.endsWith('/' + known) || known.endsWith('/' + path)) return true
  }
  return false
}

function buildSpec(source) {
  if (source === null || typeof source !== 'object') return null
  return {
    goal: source.goal ?? '',
    acceptance: Array.isArray(source.acceptance) ? source.acceptance : [],
    constraints: Array.isArray(source.constraints) ? source.constraints : [],
    readings: Array.isArray(source.readings) ? source.readings : [],
    assumptions: Array.isArray(source.assumptions) ? source.assumptions : [],
  }
}

    terms: clarify !== null && Array.isArray(clarify.terms) ? clarify.terms : [],
    questions,
    deferred_questions: deferredQuestions.map((question) => question.id),
    draft_spec: clarified,
    default_assumptions: defaultAssumptions,
    next:
      questions.length === 0
        ? 'P1 没有发现需要人拍板的点，可以直接用 draft_spec 进 stage=design。'
        : `用 ask_user_question 一次问完这 ${questions.length} 道题；`
          + '如果不方便问，直接进 stage=design，默认值会被记为假设并在结果里标出待确认项。',
  }
}

// ═══════════════════════════════════════════════════════════════════
// stage=design / all：P2 -> P3 -> P4
// ═══════════════════════════════════════════════════════════════════
// stage=design 时 spec 来自 args（人已决议）；stage=all 时用刚澄清的。
const incomingSpec = args.spec ?? null
const spec = incomingSpec !== null ? buildSpec(incomingSpec) : clarified
const acceptance = spec === null ? [] : spec.acceptance
const readings =
  spec !== null && Array.isArray(spec.readings) && spec.readings.length > 0
    ? spec.readings
    : [{ id: 'r1', summary: '', target_files: [] }]

// 已被假设覆盖的点，评审不该再报 unknown（P3 prompt 里也会说一遍）。
const assumptionText = (spec === null ? [] : spec.assumptions)
  .map((item) => item.statement)
  .join('\n')

phase('P2 强制发散')
log(`按 ${readings.length} 个解读 × ${perspectives.length} 个视角并行出方案`)

const planItems = []
for (const reading of readings) {
  for (let i = 0; i < perspectives.length; i += 1) {
    planItems.push({ reading, perspective: perspectives[i], index: i })
  }
}

const designs = await parallel(
  planItems.map((item) => () =>
    agent(
      [
        `你是方案设计者。视角：${item.perspective}`,
        '',
        `需求：${requirement}`,
        '',
        item.reading.summary === ''
          ? '（需求只有一种理解）'
          : `当前解读（${item.reading.id}）：${item.reading.summary}`
            + (item.reading.target_files.length > 0
              ? `；涉及文件：${item.reading.target_files.join('、')}`
              : ''),
        '',
        `已澄清的验收条件：${JSON.stringify(acceptance)}`,
        '',
        '已知的假设（不要重复提问，按这些假设设计）：',
        assumptionText === '' ? '（无）' : assumptionText,
        '',
        '项目里已有的资产（必须认真读完，别凭空设计）：',
        inventory,
        '',
        '规则：',
        '1. 先判断这个需求能不能直接用清单里的既有资产满足；能就直接说怎么用；',
        '2. reuses 里每条都要写 `路径:行号 — 用途`，路径必须逐字复制上面的清单，'
          + '脚本会核对，编造的路径会让整个方案作废；',
        '3. structural_choice 用「维度=取值」写清你的关键结构选择，'
          + '其它视角的方案会选别的取值，重复了说明你没有真正发散；',
        '4. second_option 必须是同一视角下真实不同的做法，不是同一方案的措辞变体；',
        '5. 如果需求里有你认为必须由人拍板的地方，把它写进 risks，不要自己拍板。',
      ].join('\n'),
      {
        schema: DESIGN_SCHEMA,
        label: item.reading.summary === ''
          ? `P2-方案${item.index + 1}`
          : `P2-${item.reading.id}-方案${item.index + 1}`,
      },
    ),
  ),
)
const validDesigns = designs.filter(Boolean)
if (validDesigns.length === 0) {
  log('所有方案生成失败，协议中止')
  return { ok: false, reason: 'no-design', spec, terms: clarify === null ? [] : clarify.terms }
}
log(`产出 ${validDesigns.length} 个方案`)

phase('P3 对抗评审')
const reviews = await parallel(
  validDesigns.map((design) => () =>
    agent(
      [
        '你是对抗评审。你的任务不是夸它，是找它会出错的地方。',
        '',
        `需求：${requirement}`,
        '',
        `方案：${JSON.stringify(design)}`,
        '',
        '项目里已有的资产：',
        inventory,
        '',
        '已知的假设（这些点已经被人或默认值定了，不要因为「假设可能不对」就报 unknown）：',
        assumptionText === '' ? '（无）' : assumptionText,
        '',
        '必须回答：',
        '1. 方案有没有放着清单里的现成资产不用、自己重新实现？reuse_verdict 必须诚实给出；'
          + '清单里确实一条都用不上时写 no-applicable-asset，并在 reuse_reason 说明为什么用不上 —— '
          + '不要为了显得有用而硬凑资产；',
        '2. reuse_reason 要具体到「哪条资产、为什么不用/怎么用」，不能只写「已考虑」；',
        '3. evidence_refs 至少引用一条清单里的 `path:line` 作为判断依据，没有依据就不要下判断；',
        '4. veto 只在方案确实不可行时投 true（会出错、违反硬约束），「我不喜欢」不算；',
        '5. unknowns 每条必须标 kind：',
        '   requirement = 需求本身没说清、只有人能定的；',
        '   evidence = 缺资料或凭据、查不到所以判断不了。',
        '   已经在上面假设清单里的点不要再报——那样只会让流程原地打转；',
        '6. issues 里每条都要指向方案里的具体表述，不要泛泛而谈。',
      ].join('\n'),
      { schema: REVIEW_SCHEMA, label: `P3-评审-${design.name}` },
    ),
  ),
)
const validReviews = reviews.filter(Boolean)

phase('P4 收敛裁决')
// 确定性硬门：这些检查不依赖模型判断，脚本自己算。
const checks = []
for (const design of validDesigns) {
  const review = validReviews.find((item) => item.design === design.name)

  // 复用证据核对：编造的路径不算证据。
  const reusePaths = design.reuses.map(reusePathOf)
  const fabricated = reusePaths.filter((path) => path === null || !isKnownPath(path))
  const verifiedReuses = reusePaths.length - fabricated.length

  const veto = review !== undefined && review.veto === true
  const rawUnknowns = review === undefined ? [] : review.unknowns
  const unknowns = rawUnknowns.filter(
    (item) => item !== null && typeof item === 'object' && typeof item.text === 'string',
  )
  const requirementUnknowns = unknowns.filter((item) => item.kind === 'requirement')
  const evidenceUnknowns = unknowns.filter((item) => item.kind !== 'requirement')

  const evidenceRefs = review === undefined ? [] : review.evidence_refs
  const evidenceGiven = evidenceRefs.some((ref) => {
    const match = String(ref).match(/^(.+?):\d+/)
    return match !== null && isKnownPath(match[1])
  })

  // 复用证据：要么真的核验通过，要么诚实说明「清单里没有可用资产」。
  // 只认 reuses 非空会惩罚诚实 —— 模型学会的是硬凑一条资产来过关，不是真的去查清单。
  const reuseVerdict = review === undefined ? 'missing' : review.reuse_verdict
  const reuseReasonLen = (review === undefined ? '' : review.reuse_reason).trim().length
  const claimsNoAsset = reuseVerdict === 'no-applicable-asset'
  // 评审口径与实际复用清单对不上：说清单用不上却列了复用项，或者说复用了却一条都没列。
  const reuseContradiction =
    (claimsNoAsset && design.reuses.length > 0)
    || (!claimsNoAsset
      && (reuseVerdict === 'fully-reused' || reuseVerdict === 'partially-reused')
      && design.reuses.length === 0)
  const reuseJustified =
    !reuseContradiction
    && (verifiedReuses > 0
      || (claimsNoAsset && design.reuses.length === 0 && reuseReasonLen >= 30))

  // 三态裁决：只缺需求信息不该算「做不出来」，它是「等人拍板」。
  const hardBlocked =
    review === undefined || veto || evidenceUnknowns.length > 0 || !evidenceGiven || fabricated.length > 0
  let decision
  const blockedBy = []
  if (hardBlocked) {
    decision = 'rejected'
    if (review === undefined) blockedBy.push('no-review')
    if (veto) blockedBy.push('veto')
    if (evidenceUnknowns.length > 0) blockedBy.push('evidence-unknowns')
    if (!evidenceGiven) blockedBy.push('no-evidence-refs')
    if (fabricated.length > 0) blockedBy.push('fabricated-reuse-path')
  } else if (requirementUnknowns.length > 0) {
    decision = 'blocked-on-human'
  } else {
    decision = 'admitted'
  }
  // 需求缺口单独补记：硬门优先是对的（缺证据不该放行），但若只写「缺证据」，
  // 人就看不到这些方案还各带着一条只有他能回答的问题，下一轮又得重挖一遍。
  // 某项目 那轮就是如此：3 个方案全部因 evidence-unknowns 被拒，同时各自还有 1–2 条需求缺口被埋掉。
  if (requirementUnknowns.length > 0) blockedBy.push('requirement-unknowns')

  checks.push({
    design: design.name,
    structural_choice: design.structural_choice,
    reuse_entries: design.reuses.length,
    reuse_verified: verifiedReuses,
    reuse_fabricated: fabricated,
    has_second: design.second_option.trim().length > 0,
    reviewed: review !== undefined,
    verdict: review === undefined ? 'missing' : review.verdict,
    veto,
    unknowns,
    requirement_unknowns: requirementUnknowns,
    evidence_unknowns: evidenceUnknowns,
    reuse_verdict: reuseVerdict,
    reuse_reason_len: reuseReasonLen,
    reuse_justified: reuseJustified,
    reuse_contradiction: reuseContradiction,
    evidence_given: evidenceGiven,
    ignored_list: review !== undefined && review.reuse_verdict === 'ignored-list',
    decision,
    // 卡住的具体原因，方便人一眼看出卡在哪
    blocked_by: blockedBy,
  })
}

// 方案去重：只看结构选择枚举是否两两唯一。
// 复用集合重叠度不参与判定 —— 清单贫瘠时（例如清单里只有样式 token、没有真正相关的组件），
// 所有方案只能复用同样那几条，Jaccard=1 反映的是清单的问题，不是方案没有发散。
// 某项目 那轮实测：三个方案结构分别是「单行事件绑定」「复用既有提交入口」「预索引+防抖」，
// 但因为清单里没有树/过滤资产，三家都只引了同样三条样式变量，重叠度被判 1.0 而误判不够发散。
// 重叠度仍保留为 reuse_overlap_max 诊断字段，供人判断是不是真撞了。
const structuralChoices = validDesigns.map((design) => normalizePath(design.structural_choice))
const structuralDistinct = new Set(structuralChoices).size === validDesigns.length

const reuseSets = validDesigns.map(
  (design) => new Set(design.reuses.map(reusePathOf).filter((path) => path !== null)),
)
let maxOverlap = 0
for (let i = 0; i < reuseSets.length; i += 1) {
  for (let j = i + 1; j < reuseSets.length; j += 1) {
    const a = reuseSets[i]
    const b = reuseSets[j]
    let shared = 0
    for (const item of a) if (b.has(item)) shared += 1
    const union = a.size + b.size - shared
    const similarity = union === 0 ? 0 : shared / union
    if (similarity > maxOverlap) maxOverlap = similarity
  }
}

/**
 * 把多个评审反复提到的需求歧义聚成一类。
 * 文本不会逐字相同，用字符二元组的 Jaccard 近似判重；被 ≥2 个评审提出的才算「一致」。
 */
function bigrams(text) {
  const clean = String(text).replace(/[\s，。、：；！？（）()「」『』"'`,.:;!?]/g, '')
  const grams = new Set()
  for (let i = 0; i < clean.length - 1; i += 1) grams.add(clean.slice(i, i + 2))
  return grams
}
function similarText(a, b) {
  const ga = bigrams(a)
  const gb = bigrams(b)
  if (ga.size === 0 || gb.size === 0) return 0
  let shared = 0
  for (const item of ga) if (gb.has(item)) shared += 1
  return shared / (ga.size + gb.size - shared)
}

const clusters = []
for (const check of checks) {
  for (const item of check.requirement_unknowns) {
    const hit = clusters.find((cluster) => similarText(cluster.text, item.text) >= 0.5)
    if (hit === undefined) {
      clusters.push({ text: item.text, sources: [check.design] })
    } else if (!hit.sources.includes(check.design)) {
      hit.sources.push(check.design)
    }
  }
}
// 被两个以上评审独立提出 = 大概率是真的问题，直接转成下一轮要问人的题。
const consistentUnknowns = clusters.filter((cluster) => cluster.sources.length >= 2)

const admittedCount = checks.filter((check) => check.decision === 'admitted').length
const blockedCount = checks.filter((check) => check.decision === 'blocked-on-human').length
const rejectedCount = checks.filter((check) => check.decision === 'rejected').length

// 有待确认假设时，结果要显式标出来，不能让人以为已经全部敲定。
const pendingAssumptions = spec === null ? [] : spec.assumptions
const underAssumptions = admittedCount > 0 && pendingAssumptions.length > 0

const levelCounts = { L0: 0, L1: 0, L2: 0, L3: 0 }
for (const item of acceptance) {
  if (item !== null && typeof item === 'object' && Object.hasOwn(levelCounts, item.level)) {
    levelCounts[item.level] += 1
  }
}

const verdictSummary =
  admittedCount > 0
    ? underAssumptions
      ? 'admitted-under-assumptions'
      : 'admitted'
    : blockedCount > 0
      ? 'blocked-on-human'
      : 'rejected'

const reqGapCount = checks.filter((check) => check.requirement_unknowns.length > 0).length
let next = ''
if (admittedCount > 0 && underAssumptions) {
  next = `有 ${admittedCount} 个方案可以推进，但建立在 ${pendingAssumptions.length} 条未确认的假设上。`
    + '先看 pending_assumptions，确认没问题再开工；任何一条不成立，对应方案要重做。'
} else if (admittedCount > 0) {
  next = '由人结合 gate 与 checks 里 decision=admitted 的方案选择；'
    + 'any_ignored_existing_assets 为真时，先追问为什么不用现成资产。'
} else if (blockedCount > 0) {
  next = `没有方案可以开工，但缺的是需求信息不是能力（${blockedCount} 个方案只差人拍板）。`
    + '把 next_round_questions 拿去问人，答完用 stage=design 带新 spec 重跑，不要重新发散。'
    + '这一步记入失败库，分类为「需求误解」。'
} else {
  next = '所有方案都被否决或证据不足。先看 checks 里的 blocked_by 定位原因，'
    + '再决定是补证据还是换思路；不要原样重跑。'
  // 硬门优先是对的，但若只报告「缺证据」，人看不到这些方案还各有只有他能回答的问题。
  if (reqGapCount > 0) {
    next += `另外其中 ${reqGapCount} 个方案还带着需求缺口（blocked_by 里的 requirement-unknowns），`
      + '补证据的同时把这些点一起拿去问人，别等下一轮又卡在同一个地方。'
  }
}

return {
  ok: true,
  stage: 'design',
  requirement,
  terms: clarify !== null && Array.isArray(clarify.terms) ? clarify.terms : [],
  spec,
  acceptance_levels: levelCounts,
  designs: validDesigns,
  reviews: validReviews,
  checks,
  pending_assumptions: pendingAssumptions,
  consistent_unknowns: consistentUnknowns,
  // 下一轮要问人的题；blocked-on-human 时直接拿去用，不要重新发散。
  next_round_questions: consistentUnknowns.map((cluster, index) => ({
    id: `q${index + 1}`,
    kind: 'requirement',
    question: cluster.text,
    raised_by: cluster.sources,
  })),
  verdict: verdictSummary,
  gate: {
    designs_expected: planItems.length,
    designs_produced: validDesigns.length,
    designs_admitted: admittedCount,
    designs_blocked_on_human: blockedCount,
    designs_rejected: rejectedCount,
    // 「有据可依」有两种合法形态：核验过的复用，或诚实说明清单里没有可用资产。
    all_have_reuse_evidence: checks.every((check) => check.reuse_justified),
    all_reuse_paths_verified: checks.every((check) => check.reuse_fabricated.length === 0),
    all_have_second_option: checks.every((check) => check.has_second),
    all_reviewed: checks.every((check) => check.reviewed),
    all_evidence_backed: checks.every((check) => check.evidence_given),
    designs_are_distinct: structuralDistinct,
    designs_with_requirement_gaps: reqGapCount,
    structural_choices_distinct: structuralDistinct,
    reuse_overlap_max: Math.round(maxOverlap * 100) / 100,
    any_ignored_existing_assets: checks.some((check) => check.ignored_list),
    any_veto: checks.some((check) => check.veto),
    any_unknowns: checks.some((check) => check.unknowns.length > 0),
    any_requirement_unknowns: checks.some((check) => check.requirement_unknowns.length > 0),
    any_evidence_unknowns: checks.some((check) => check.evidence_unknowns.length > 0),
    any_fabricated_reuse: checks.some((check) => check.reuse_fabricated.length > 0),
    any_reuse_contradiction: checks.some((check) => check.reuse_contradiction),
    under_assumptions: underAssumptions,
  },
  // 脚本不做「哪个方案最好」的判断 —— 那需要人拍板。
  next,
}
